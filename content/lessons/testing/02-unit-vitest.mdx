  <h2>Базова анатомія тесту: патерн AAA (Arrange - Act - Assert)</h2>

```typescript
// utils/formatCurrency.ts
export function formatCurrency(amount: number, currency: 'UAH' | 'USD' = 'UAH'): string {
  if (isNaN(amount)) throw new Error('Невалідне число');
  return new Intl.NumberFormat('uk-UA', {
    style: 'currency',
    currency,
    maximumFractionDigits: 0,
  }).format(amount);
}
```

```typescript
// utils/formatCurrency.test.ts
import { describe, it, expect } from 'vitest';
import { formatCurrency } from './formatCurrency';

describe('formatCurrency()', () => {
  it('коректно форматує суму в гривнях', () => {
    // 1. Arrange (Підготовка даних)
    const input = 1500;

    // 2. Act (Виконання функції)
    const result = formatCurrency(input, 'UAH');

    // 3. Assert (Перевірка очікуваного результату)
    expect(result).toContain('1');
    expect(result).toContain('500');
    expect(result).toContain('грн');
  });

  it('викидає помилку при передачі NaN', () => {
    expect(() => formatCurrency(NaN)).toThrow('Невалідне число');
  });
});
```

  <h2>Параметризовані тести з `test.each`</h2>
  <p>Замість дублювання коду тесту для різних вхідних даних — використовуйте табличні тести:</p>

```typescript
describe('перевірка валідності email', () => {
  it.each([
    ['test@example.com', true],
    ['user.name+tag@domain.co.uk', true],
    ['invalid-email', false],
    ['@missing-user.com', false],
    ['spaces in@email.com', false],
  ])('для вводу %s повертає %s', (email, expected) => {
    expect(isValidEmail(email)).toBe(expected);
  });
});
```
