  <h2>1. Форматування чисел та валют: Intl.NumberFormat</h2>

```javascript
// 1. Форматування цін у гривнях (UAH)
const uahFormatter = new Intl.NumberFormat('uk-UA', {
  style: 'currency',
  currency: 'UAH',
});
console.log(uahFormatter.format(124500.75)); // "124 500,75 ₴"

// 2. Форматування цін у доларах (USD)
const usdFormatter = new Intl.NumberFormat('en-US', {
  style: 'currency',
  currency: 'USD',
});
console.log(usdFormatter.format(124500.75)); // "$124,500.75"

// 3. Компактні великі числа для лічильників переглядів (1.2M, 50K)
const compactFormatter = new Intl.NumberFormat('uk-UA', {
  notation: 'compact',
  compactDisplay: 'short',
});
console.log(compactFormatter.format(1_500_000)); // "1,5 млн"
```

  <h2>2. Форматування дат і часу: Intl.DateTimeFormat</h2>

```javascript
const now = new Date('2026-09-24T18:30:00');

// Повний текстовий формат українською
const ukDate = new Intl.DateTimeFormat('uk-UA', {
  dateStyle: 'full',
  timeStyle: 'short',
}).format(now);
console.log(ukDate); // "четвер, 24 вересня 2026 р. о 18:30"
```

  <h2>3. Відносний час: Intl.RelativeTimeFormat</h2>

```javascript
const rtf = new Intl.RelativeTimeFormat('uk-UA', { numeric: 'auto' });

console.log(rtf.format(-1, 'day'));    // "вчора"
console.log(rtf.format(-5, 'minute')); // "5 хвилин тому"
console.log(rtf.format(2, 'month'));   // "через 2 місяці"
```

  <h2>4. Граматика числівників (плюралізація): Intl.PluralRules</h2>

```javascript
// В українській мові є 3 форми множини: one (1), few (2-4), many (5-20)
const pr = new Intl.PluralRules('uk-UA');

const suffixes = {
  one: 'урок',
  few: 'уроки',
  many: 'уроків',
  other: 'уроків',
};

function formatLessonsCount(count) {
  const rule = pr.select(count); // повертає 'one' | 'few' | 'many'
  return `${count} ${suffixes[rule]}`;
}

console.log(formatLessonsCount(1));  // "1 урок"
console.log(formatLessonsCount(3));  // "3 уроки"
console.log(formatLessonsCount(15)); // "15 уроків"
console.log(formatLessonsCount(21)); // "21 урок"
```
