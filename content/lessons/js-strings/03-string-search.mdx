  <h2>1. Перевірка наявності: str.includes(substring, startPos)</h2>

```javascript
const sentence = "JavaScript — мова вебу та інтерактивності";

console.log(sentence.includes("мова")); // true
console.log(sentence.includes("Python")); // false

// Опціональний другий параметр — позиція, з якої починати пошук:
console.log(sentence.includes("JavaScript", 5)); // false (бо починаємо з 5 індексу)
```

  <h2>2. Пошук позиції: str.indexOf та str.lastIndexOf</h2>

```javascript
const text = "Widget with id and name widget";

// indexOf повертає індекс першого входження або -1, якщо не знайдено
console.log(text.indexOf("with")); // 7
console.log(text.indexOf("not_found")); // -1

// Пошук з певного індексу
console.log(text.indexOf("widget", 10)); // 24 (знайшов друге слово)

// lastIndexOf шукає з кінця рядка до початку
console.log(text.lastIndexOf("widget")); // 24
```

  <Callout type="warning" title="Класична пастка: if (str.indexOf(...))">
    Ніколи не пишіть умову так:
    ```javascript
    // ❌ ПОМИЛКА: якщо підрядок на самому початку (індекс 0), 0 перетвориться на false!
    if (str.indexOf("JavaScript")) { ... }

    // ✅ ПРАВИЛЬНО:
    if (str.includes("JavaScript")) { ... }
    // або:
    if (str.indexOf("JavaScript") !== -1) { ... }
    ```
  </Callout>

  <h2>3. Початок та кінець рядка: startsWith та endsWith</h2>

```javascript
const url = "https://example.com/assets/logo.svg";

// startsWith перевіряє початок
console.log(url.startsWith("https://")); // true
console.log(url.startsWith("http://"));  // false

// endsWith перевіряє закінчення
console.log(url.endsWith(".svg")); // true
console.log(url.endsWith(".png")); // false

// endsWith приймає другий аргумент — максимальну довжину для перевірки:
console.log(url.endsWith(".com", 19)); // true (перевіряє рядок до 19 символу: "https://example.com")
```

  <h2>Регістронезалежний пошук (Case-Insensitive Search)</h2>

```javascript
function containsWord(source, query) {
  return source.toLowerCase().includes(query.trim().toLowerCase());
}

console.log(containsWord("React & TypeScript Camp", "typescript")); // true
console.log(containsWord("Next.js 15 App", "NEXT")); // true
```

  if (typeof url !== 'string') return false;
  return url.startsWith('https://') || url.startsWith('wss://');
}

export function countOccurrences(text, word) {
  if (!text || !word) return 0;
  const lowerText = text.toLowerCase();
  const lowerWord = word.toLowerCase();
  let count = 0;
  let pos = 0;

  while ((pos = lowerText.indexOf(lowerWord, pos)) !== -1) {
    count++;
    pos += lowerWord.length;
  }
  return count;
}

export function highlightKeyword(text, keyword) {
  if (!text || !keyword) return text;
  const regex = new RegExp(\`(\${keyword.replace(/[-[\\]{}()*+?.,\\\\^$|#\\s]/g, '\\\\$&')})\`, 'gi');
  return text.replace(regex, '<mark>$1</mark>');
}`}
