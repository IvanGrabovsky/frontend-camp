```javascript
class TextProcessor {
  constructor(rawText = '') {
    this.text = rawText.trim();
  }

  // 1. Статистика слів та символів
  getStats() {
    const characters = this.text.length;
    const charactersWithoutSpaces = this.text.replace(/\s+/g, '').length;
    const words = this.text ? this.text.match(/[\wа-яіїєґ'-]+/gi)?.length || 0 : 0;
    const readingTimeMinutes = Math.max(1, Math.ceil(words / 200));

    return {
      characters,
      charactersWithoutSpaces,
      words,
      readingTimeMinutes,
    };
  }

  // 2. Генерація безпечного HTML прев'ю
  toSafeHtml() {
    return this.text
      .replaceAll('&', '&amp;')
      .replaceAll('<', '&lt;')
      .replaceAll('>', '&gt;')
      .replace(/\*\*(.*?)\*\*/g, '<strong>$1</strong>')
      .replace(/\*(.*?)\*/g, '<em>$1</em>')
      .replace(/`([^`]+)`/g, '<code>$1</code>');
  }
}
```
