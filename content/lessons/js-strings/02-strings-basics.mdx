  <h2>Довжина рядка та доступ до символів</h2>

```javascript
const course = "Frontend Camp";

// Властивість length повертає кількість UTF-16 кодових одиниць
console.log(course.length); // 13

// 1. Доступ через квадратні дужки (квадратні дужки не підтримують від'ємні індекси)
console.log(course[0]); // "F"
console.log(course[course.length - 1]); // "p" (останній символ)

// 2. Сучасний метод .at() (підтримує від'ємні індекси від кінця!)
console.log(course.at(0));  // "F"
console.log(course.at(-1)); // "p" (останній символ)
console.log(course.at(-2)); // "m" (передостанній символ)
```

  <Callout type="info" title="Чому метод .at() зручніший за дужки []?">
    Раніше, щоб дістати останній символ, доводилося писати довгу конструкцію <code>str[str.length - 1]</code>. Метод <code>str.at(-1)</code> значно лаконічніший і працює однаково зручно як з рядками, так і з масивами.
  </Callout>

  <h2>Зміна регістру: toUpperCase та toLowerCase</h2>

```javascript
const email = "  User.Name@Example.COM  ";

// Приведення до верхнього та нижнього регістру
console.log(email.toUpperCase()); // "  USER.NAME@EXAMPLE.COM  "
console.log(email.toLowerCase()); // "  user.name@example.com  "

// Локалізоване порівняння (важливо для турецької, німецької та ін.)
const city = "KYIV";
console.log(city.toLocaleLowerCase('uk-UA')); // "kyiv"
```

  <h2>Очищення пробілів: trim, trimStart, trimEnd</h2>

```javascript
const rawInput = "   Привіт, Світе! \n\t  ";

// trim() видаляє всі невидимі символи з обох кінців
const cleanInput = rawInput.trim();
console.log(`"${cleanInput}"`); // "Привіт, Світе!"

// Очищення тільки з одного боку:
console.log(rawInput.trimStart()); // прибирає пробіли лише на початку
console.log(rawInput.trimEnd());   // прибирає пробіли лише в кінці
```

  <h2>Доповнення довжини: padStart та padEnd</h2>

```javascript
// 1. Форматування номерів із лідируючими нулями (наприклад: 001, 002, 042)
const id = "42";
console.log(id.padStart(5, "0")); // "00042"

// 2. Маскування номеру банківської картки
const cardNumber = "4111222233339876";
const last4Digits = cardNumber.slice(-4);
const maskedCard = last4Digits.padStart(cardNumber.length, "*");
console.log(maskedCard); // "************9876"

// 3. Табличне вирівнювання з padEnd
console.log("Товар".padEnd(15, ".") + "120 грн"); // "Товар..........120 грн"
console.log("Доставка".padEnd(15, ".") + "50 грн");  // "Доставка.......50 грн"
```

  <h2>Повторення рядків: str.repeat(n)</h2>

```javascript
console.log("★".repeat(5)); // "★★★★★"
console.log("Loading" + ".".repeat(3)); // "Loading..."
```

  if (!str) return '';

export function maskCardNumber(cardNumber, visibleDigits = 4) {
  const clean = String(cardNumber).replace(/\\s+/g, '');
