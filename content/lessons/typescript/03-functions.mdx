function calculateTotal(price: number, quantity: number = 1, discount?: number): number {
  const base = price * quantity;
  return discount ? base * (1 - discount) : base;
}

// Типізація функції як значення (Function Type Signature)

  <h2>2. Звуження типів (Type Narrowing)</h2>
  <p>
    Коли змінна має тип <code>union</code> (наприклад, <code>string | number</code>), компілятор не дозволить викликати методи, специфічні для одного з типів, доки ви не зробите звуження:
  </p>

  <ul>
    <li><code>typeof</code> — для примітивів (<code>"string"</code>, <code>"number"</code>, <code>"boolean"</code>, <code>"symbol"</code>, <code>"undefined"</code>).</li>
    <li><code>instanceof</code> — для екземплярів класів або вбудованих об'єктів (<code>Date</code>, <code>Error</code>, <code>HTMLElement</code>).</li>
    <li><code>"prop" in obj</code> — перевірка наявності ключа в об'єкті.</li>
    <li><strong>Custom Type Guards</strong> з предикатом <code>value is TargetType</code>.</li>
  </ul>

  <h2>3. Custom Type Guards (`is`)</h2>
  <p>
    Коли логіка перевірки складна, ми виносимо її у спеціальну функцію-предикат:
  </p>

```typescript
interface ApiResponseSuccess {
  status: "success";
  data: { id: string; name: string };
}

interface ApiResponseError {
  status: "error";
  message: string;
}

type ApiResponse = ApiResponseSuccess | ApiResponseError;

// Custom Type Guard з предикатом 'is'
function isSuccessResponse(res: ApiResponse): res is ApiResponseSuccess {
  return res.status === "success";
}

function handleResponse(res: ApiResponse) {
  if (isSuccessResponse(res)) {
    console.log("Отримано користувача:", res.data.name);
  } else {
    console.error("Помилка запиту:", res.message);
  }
}
```
